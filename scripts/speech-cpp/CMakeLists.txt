cmake_minimum_required(VERSION 3.16)
project(basqar_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES ON CACHE BOOL "" FORCE)
if(NOT WHISPER_SOURCE_DIR)
  message(FATAL_ERROR "WHISPER_SOURCE_DIR must reference the pinned whisper.cpp source")
endif()
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper)
add_executable(basqar-speech runner.cpp)
target_link_libraries(basqar-speech PRIVATE whisper)
